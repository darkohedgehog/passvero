import hashlib, importlib.util, json, os, pathlib, stat
P=pathlib.Path
O=P('/var/lib/passvero-signature-continuity-20261021-01/rollback')
H=P('/var/lib/passvero-auth-flow-continuation-20261020-01/recover-release-reviewed.py')
ID='recovery-20261021-continuity-01'
result={'automaticRetry':False,'attempt':ID}
try:
    st=H.lstat()
    assert stat.S_ISREG(st.st_mode) and st.st_uid==0 and not st.st_mode & 0o022 and st.st_nlink==1
    assert hashlib.sha256(H.read_bytes()).hexdigest()=='0e3938d06d6bcbfa9eab0249479bab5e54fe5162a3669a3026a39a0d204c4f44'
    spec=importlib.util.spec_from_file_location('accepted_recovery',H)
    h=importlib.util.module_from_spec(spec); spec.loader.exec_module(h)
    with h.exclusive(h.RECORDS/'operator.lock'):
        host=h.LinuxHost(ID)
        state=h.read_state(host.archive/'state.json')
        if state is None:
            result['closure']='NO_RECOVERY_PHASE_STARTED_NO_SERVICE_ACTION'
        else:
            if state.get('phase')=='ROLLED_BACK':
                h.require(h.active(h.TIMER) in ('inactive','failed'),'ROLLBACK_TIMER_NOT_DISABLED')
                value=h.run(['/usr/bin/systemctl','is-enabled',h.TIMER],success=False)
                h.require(value.stdout.strip()==b'disabled','ROLLBACK_TIMER_NOT_DISABLED')
                result['closure']='PASS_EXPLICIT_ROLLBACK_PRODUCER_DISABLED_TRUST_NOT_RESTORED'
            else:
                host.contain()
                result['closure']='FAIL_CLOSED_ROLLBACK_INCOMPLETE_OPERATOR_REVIEW'
                result['phase']=state.get('phase','UNKNOWN')
    h.write_state(O/'closure.json',result)
except BaseException as error:
    result.update(closure='PARTIAL_CONTAINMENT_REQUIRES_OPERATOR_REVIEW',reason=type(error).__name__)
    fd=os.open(O/'closure-failure.json',os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
    with os.fdopen(fd,'w') as stream: json.dump(result,stream);stream.flush();os.fsync(stream.fileno())
    raise SystemExit(1)
